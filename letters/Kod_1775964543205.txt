            headers: API_HEADERS,
            cache: 'no-store'
        });
        
        if (!response.ok) {
            throw new Error(`GitHub API xətası: ${response.status}`);
        }

                        headers: API_HEADERS
                    });

                        const commits = await commitsResponse.json();

                            file.commit_date = commits[0].commit.author.date;

        );

                .filter(f => /\.(txt|md)$/i.test(f.name))
                .sort((a, b) => {

        showError('Məktublar yüklənə bilmədi');
        showLettersEmpty();
    } finally {
        AppState.isLoading.letters = false;
    }
